# Krishi Seva Kendra — Node.js Backend Dependencies
# This is a Node.js project. Install dependencies using:
#   npm install

express-validator@^7.1.0   # Request validation middleware
jsonwebtoken@^9.0.2        # JWT authentication tokens
mongoose@^8.24.4           # MongoDB ODM (Object Document Mapper)
